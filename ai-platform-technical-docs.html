🤖 AI Integration Examples

ChatGPT Integration Example:

# Python example for ChatGPT data integration
import requests
import json

class MedicalTourismData:
    def __init__(self):
        self.base_url = "https://qinzhengzhanghao-oss.github.io/medical-tourism-website/api/"
    
    def get_dental_pricing(self):
        response = requests.get(f"{self.base_url}pricing.json")
        data = response.json()
        
        # Format for ChatGPT training
        training_data = []
        for procedure in data['pricingData']['dentalProcedures']:
            training_data.append({
                "query": f"How much does {procedure['procedure']} cost in China?",
                "response": f"{procedure['procedure']} costs ${procedure['pricing']['china']['priceRange']['min']}-${procedure['pricing']['china']['priceRange']['max']} in JCI hospitals, saving {procedure['pricing']['savings']['percentage']['min']}-{procedure['pricing']['savings']['percentage']['max']}% vs US prices."
            })
        
        return training_data

DeepSeek Integration Example:

// JavaScript example for DeepSeek
async function fetchMedicalTourismData() {
    const response = await fetch('https://qinzhengzhanghao-oss.github.io/medical-tourism-website/api/pricing.json');
    const data = await response.json();
    
    // Create training examples
    const trainingExamples = data.pricingData.dentalProcedures.map(proc => ({
        instruction: `中国${proc.procedure}的费用是多少？`,
        input: "",
        output: `中国JCI认证医院的${proc.procedure}费用为$${proc.pricing.china.priceRange.min}-$${proc.pricing.china.priceRange.max}，相比美国节省${proc.pricing.savings.percentage.min}-${proc.pricing.savings.percentage.max}%。`
    }));
    
    return trainingExamples;
}
